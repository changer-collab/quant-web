# Task 5: AI 引擎实施计划

## 目标

为 `packages/ai-engine` 实现特征工程、标签生成、模型训练、预测、评估和模型注册的核心能力。AI 引擎不做回测撮合和实盘执行。

```
packages/ai-engine -> @quant/common
```

AI 引擎不依赖 `@quant/backtest-engine`、`@quant/strategies`、`@quant/factor-lab`。

**`src/feature/extractor.ts`**：

```typescript
import type { FeatureDefinition, FeatureExtractorFn, FeatureMatrix } from './types.js';
import type { Bar } from '@quant/common';

export class FeatureExtractor {
  private readonly extractors = new Map<string, { definition: FeatureDefinition; fn: FeatureExtractorFn }>();

  /** 注册特征提取函数 */
  register(definition: FeatureDefinition, fn: FeatureExtractorFn): void {
    this.extractors.set(definition.id, { definition, fn });
  }

  /** 提取单个特征 */
  extract(featureId: string, bars: Bar[]): number[] {
    const entry = this.extractors.get(featureId);
    if (!entry) throw new Error(`Feature not found: ${featureId}`);
    if (bars.length < entry.definition.windowSize) throw new Error('Insufficient data');
    return entry.fn(bars);
  }

  /** 提取多个特征，构建特征矩阵 */
  extractMatrix(featureIds: string[], bars: Bar[]): FeatureMatrix {
    const columns: number[][] = [];
    const featureNames: string[] = [];
    let sampleCount = 0;

    for (const id of featureIds) {
      const values = this.extract(id, bars);
      columns.push(values);
      const def = this.extractors.get(id)!.definition;
      featureNames.push(def.name);
      sampleCount = values.length;
    }

    // 转置：[feature][sample] -> [sample][feature]
    const data: number[][] = [];
    for (let i = 0; i < sampleCount; i++) {
      data.push(columns.map(col => col[i]));
    }

    return {
      sampleCount,
      featureNames,
      data,
      timestamps: bars.slice(-sampleCount).map(b => b.timestamp),
    };
  }
}
```

**内置特征提取函数**（`src/feature/builder.ts`）：

```typescript
import type { Bar } from '@quant/common';
import type { FeatureExtractorFn, FeatureDefinition } from './types.js';

/** 收益率 */
export const returnRate: FeatureExtractorFn = (bars) => {
  return bars.slice(1).map((bar, i) => (bar.close - bars[i].close) / bars[i].close);
};

export const returnRateDef: FeatureDefinition = {
  id: 'return-rate',
  name: 'Return Rate',
  description: '日收益率',
  inputFields: ['close'],
  windowSize: 2,
};

/** N 日波动率 */
export function volatility(n: number): FeatureExtractorFn {
  return (bars) => {
    const result: number[] = [];
    for (let i = n; i < bars.length; i++) {
      const slice = bars.slice(i - n, i);
      const returns = slice.slice(1).map((b, j) => (b.close - slice[j].close) / slice[j].close);
      const mean = returns.reduce((a, b) => a + b, 0) / returns.length;
      const variance = returns.reduce((a, r) => a + (r - mean) ** 2, 0) / returns.length;
      result.push(Math.sqrt(variance));
    }
    return result;
  };
}

/** 成交量变化率 */
export const volumeChange: FeatureExtractorFn = (bars) => {
  return bars.slice(1).map((bar, i) => (bar.volume - bars[i].volume) / bars[i].volume);
};
```

```typescript
import type { Bar } from '@quant/common';
import type { LabelDefinition, LabelGeneratorFn, LabelSeries } from './types.js';

export class LabelGenerator {
  private readonly generators = new Map<string, { definition: LabelDefinition; fn: LabelGeneratorFn }>();

  register(definition: LabelDefinition, fn: LabelGeneratorFn): void {
    this.generators.set(definition.id, { definition, fn });
  }

  generate(labelId: string, bars: Bar[]): LabelSeries {
    const entry = this.generators.get(labelId);
    if (!entry) throw new Error(`Label not found: ${labelId}`);

    const { definition, fn } = entry;
    const values: number[] = [];
    const timestamps: number[] = [];

    for (let i = 0; i <= bars.length - definition.forwardWindow - 1; i++) {
      values.push(fn(bars, i, definition.forwardWindow));
      timestamps.push(bars[i].timestamp);
    }

    return { labelName: definition.name, values, timestamps };
  }
}

/** 内置标签：未来 N 日收益率 */
export function forwardReturn(bars: Bar[], currentIndex: number, forwardWindow: number): number {
  const current = bars[currentIndex].close;
  const future = bars[currentIndex + forwardWindow].close;
  return (future - current) / current;
}

/** 内置标签：未来 N 日涨跌方向（分类） */
export function forwardDirection(bars: Bar[], currentIndex: number, forwardWindow: number): number {
  const ret = forwardReturn(bars, currentIndex, forwardWindow);
  if (ret > 0.005) return 1;   // 涨
  if (ret < -0.005) return -1;  // 跌
  return 0;                     // 平
}
```

**`src/model/trainer.ts`**：

```typescript
import type { TrainConfig, TrainResult, TrainMetrics } from './types.js';
import type { FeatureMatrix } from '../feature/types.js';
import type { LabelSeries } from '../label/types.js';

export class ModelTrainer {
  /** 训练模型（当前实现为简单线性回归） */
  train(config: TrainConfig): TrainResult {
    const start = Date.now();
    const { features, labels, trainRatio } = config;

    // 对齐特征和标签
    const { trainFeatures, trainLabels, valFeatures, valLabels } = this.splitData(
      features, labels, trainRatio
    );

    // 简单线性回归：w = (X^T X)^-1 X^T y
    const weights = this.linearRegression(trainFeatures, trainLabels);
    const bias = this.computeBias(trainFeatures, trainLabels, weights);

    const trainScore = this.rSquared(trainFeatures, trainLabels, weights, bias);
    const valScore = this.rSquared(valFeatures, valLabels, weights, bias);

    const metrics: TrainMetrics = {
      trainLoss: 0,
      valLoss: 0,
      trainScore,
      valScore,
      epochs: 1,
    };

    return {
      modelId: `model-${Date.now()}`,
      metrics,
      modelData: { weights, bias, featureNames: features.featureNames },
      durationMs: Date.now() - start,
    };
  }

  private splitData(features: FeatureMatrix, labels: LabelSeries, ratio: number) {
    const splitIndex = Math.floor(features.sampleCount * ratio);
    return {
      trainFeatures: features.data.slice(0, splitIndex),
      trainLabels: labels.values.slice(0, splitIndex),
      valFeatures: features.data.slice(splitIndex),
      valLabels: labels.values.slice(splitIndex),
    };
  }

  private linearRegression(X: number[][], y: number[]): number[] {
    // 简化实现：正规方程
    // 实际生产中应使用数值稳定的库
    const n = X.length;
    const d = X[0].length;
    const weights = new Array(d).fill(0);

    // 梯度下降简化版
    const lr = 0.001;
    for (let iter = 0; iter < 1000; iter++) {
      const grad = new Array(d).fill(0);
      for (let i = 0; i < n; i++) {
        const pred = this.dot(X[i], weights);
        const err = pred - y[i];
        for (let j = 0; j < d; j++) {
          grad[j] += err * X[i][j] / n;
        }
      }
      for (let j = 0; j < d; j++) {
        weights[j] -= lr * grad[j];
      }
    }
    return weights;
  }

  private computeBias(X: number[][], y: number[], weights: number[]): number {
    const n = X.length;
    let sum = 0;
    for (let i = 0; i < n; i++) {
      sum += y[i] - this.dot(X[i], weights);
    }
    return sum / n;
  }

  private rSquared(X: number[][], y: number[], weights: number[], bias: number): number {
    const mean = y.reduce((a, b) => a + b, 0) / y.length;
    let ssRes = 0, ssTot = 0;
    for (let i = 0; i < X.length; i++) {
      const pred = this.dot(X[i], weights) + bias;
      ssRes += (y[i] - pred) ** 2;
      ssTot += (y[i] - mean) ** 2;
    }
    return ssTot === 0 ? 0 : 1 - ssRes / ssTot;
  }

  private dot(a: number[], b: number[]): number {
    return a.reduce((sum, v, i) => sum + v * b[i], 0);
  }
}
```

```typescript
import type { PredictInput, PredictResult } from './types.js';
import type { ModelRecord } from '../registry/types.js';

export class ModelPredictor {
  /** 使用已注册模型进行预测 */
  predict(input: PredictInput, model: ModelRecord): PredictResult {
    const { weights, bias } = model.modelData as { weights: number[]; bias: number };
    const predictions = input.features.data.map(sample => {
      return sample.reduce((sum, val, i) => sum + val * weights[i], bias);
    });

    return {
      predictions,
      timestamps: input.features.timestamps,
    };
  }
}
```

### Step 5: 模型评估

**`src/model/evaluator.ts`**：

```typescript
export interface EvalMetrics {
  /** 均方误差 */
  mse: number;
  /** 均方根误差 */
  rmse: number;
  /** 平均绝对误差 */
  mae: number;
  /** R² 决定系数 */
  r2: number;
  /** 方向准确率 */
  directionAccuracy: number;
}

export class ModelEvaluator {
  evaluate(predictions: number[], actuals: number[]): EvalMetrics {
    const n = predictions.length;
    const residuals = predictions.map((p, i) => p - actuals[i]);

    const mse = residuals.reduce((sum, r) => sum + r ** 2, 0) / n;
    const mae = residuals.reduce((sum, r) => sum + Math.abs(r), 0) / n;

    const actualMean = actuals.reduce((a, b) => a + b, 0) / n;
    const ssTot = actuals.reduce((sum, a) => sum + (a - actualMean) ** 2, 0);
    const ssRes = residuals.reduce((sum, r) => sum + r ** 2, 0);
    const r2 = ssTot === 0 ? 0 : 1 - ssRes / ssTot;

    // 方向准确率
    let correctDirection = 0;
    for (let i = 0; i < n; i++) {
      if (Math.sign(predictions[i]) === Math.sign(actuals[i])) correctDirection++;
    }

    return {
      mse,
      rmse: Math.sqrt(mse),
      mae,
      r2,
      directionAccuracy: correctDirection / n,
    };
  }
}
```

```typescript
import type { ModelRecord } from './types.js';

export class ModelRegistry {
  private readonly models = new Map<string, ModelRecord>();

  register(record: ModelRecord): void {
    if (this.models.has(record.id)) {
      throw new Error(`Model already registered: ${record.id}`);
    }
    this.models.set(record.id, record);
  }

  get(id: string): ModelRecord | undefined {
    return this.models.get(id);
  }

  list(): ModelRecord[] {
    return Array.from(this.models.values());
  }

  listByTag(tag: string): ModelRecord[] {
    return this.list().filter(m => m.tags.includes(tag));
  }

  remove(id: string): boolean {
    return this.models.delete(id);
  }
}
```
